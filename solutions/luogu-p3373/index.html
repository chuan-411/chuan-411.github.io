已 AC P3373 · 【模板】线段树 2

2026.09.30 · 难度 3/5 · 1 分钟 #线段树#懒标记#模板
“
加法标记和乘法标记必须「先乘后加」地下发，且新乘法要同时作用在加法标记上。
—— 这道题唯一要带走的一句话

题意

维护序列，支持：

  • 区间乘 x
  • 区间加 x
  • 区间求和（对 p 取模）

标记设计

每个节点维护 (mul, add)，表示这个区间的每个数先乘 mul 再加 add。

真实值：a[i] = a[i] * mul + add。

两个标记合并的规则：

  • 新来一个「乘 m」：mul *= m，add *= m。
  • 新来一个「加 v」：add += v。

为什么乘法要作用在 add 上：原来的变换是 x → x·mul + add，再乘 m 后变成 x·mul·m + add·m，所以 add 也得乘 m。

代码

const int N = 1e5 + 5;
long long sum[N << 2], mul[N << 2], add[N << 2];
long long a[N], p;

void build(int u, int l, int r) {
    mul[u] = 1; add[u] = 0;
    if (l == r) { sum[u] = a[l] % p; return; }
    int mid = (l + r) >> 1;
    build(u << 1, l, mid);
    build(u << 1 | 1, mid + 1, r);
    sum[u] = (sum[u << 1] + sum[u << 1 | 1]) % p;
}

void apply(int u, int l, int r, long long m, long long v) {
    sum[u] = (sum[u] * m + v * (r - l + 1)) % p;
    mul[u] = mul[u] * m % p;
    add[u] = (add[u] * m + v) % p;
}

void pushdown(int u, int l, int r) {
    if (mul[u] == 1 && add[u] == 0) return;
    int mid = (l + r) >> 1;
    apply(u << 1, l, mid, mul[u], add[u]);
    apply(u << 1 | 1, mid + 1, r, mul[u], add[u]);
    mul[u] = 1; add[u] = 0;
}

apply 同时处理「乘」和「加」，pushdown 只调用一次，逻辑统一，不容易错。

查询与修改

void modify(int u, int l, int r, int ql, int qr, long long m, long long v) {
    if (ql <= l && r <= qr) { apply(u, l, r, m, v); return; }
    pushdown(u, l, r);
    int mid = (l + r) >> 1;
    if (ql <= mid) modify(u << 1, l, mid, ql, qr, m, v);
    if (qr > mid)  modify(u << 1 | 1, mid + 1, r, ql, qr, m, v);
    sum[u] = (sum[u << 1] + sum[u << 1 | 1]) % p;
}

long long query(int u, int l, int r, int ql, int qr) {
    if (ql <= l && r <= qr) return sum[u];
    pushdown(u, l, r);
    int mid = (l + r) >> 1;
    long long res = 0;
    if (ql <= mid) res = (res + query(u << 1, l, mid, ql, qr)) % p;
    if (qr > mid)  res = (res + query(u << 1 | 1, mid + 1, r, ql, qr)) % p;
    return res;
}

调用方式：乘就传 (x, 0)，加就传 (1, x)。

坑点（全是血泪）

  1. build 里 mul[u] = 1：不初始化的话，默认 0，一乘全变 0。这个错误我犯了两次。
  2. sum[u] * m 溢出：m 和 sum 都是 1e9 级别，long long 乘完再取模勉强安全（1e18），但 v * (r-l+1) 也要用 long long。
  3. p = 1 的特殊情况：所有数对 1 取模都是 0，能过，但要保证没有除零。
  4. 标记为 (1, 0) 时提前返回：不返回会一直递归，常数爆炸。

收获

模板题的价值不在「会写」，而在「写对」。标记的复合顺序是这类题唯一的思维点，其余都是手速。

写完这题之后，我看到「区间加 + 区间乘」就知道答案长什么样了。
